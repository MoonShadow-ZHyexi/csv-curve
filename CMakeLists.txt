cmake_minimum_required(VERSION 3.21)
project(CsvCurves LANGUAGES CXX)

option(CSV_CURVE_WIN7 "Build the Windows 7 edition with a Windows 7-capable MSVC toolset" OFF)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

if(MSVC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    add_compile_options("$<$<COMPILE_LANGUAGE:CXX>:/utf-8;/W4;/EHsc;/fp:precise>")
endif()

add_executable(convert convert.cpp)

if(WIN32)
    enable_language(RC)
    add_executable(csv_curve WIN32 curve_app.cpp curve_chart.cpp curve_data.cpp pwm_data.cpp curve_app.rc)
    target_include_directories(csv_curve PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    set_source_files_properties(curve_app.rc PROPERTIES OBJECT_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/curve_app.ico")
    target_compile_definitions(csv_curve PRIVATE UNICODE _UNICODE NOMINMAX)
    target_link_libraries(csv_curve PRIVATE gdiplus comdlg32 shell32 ole32 gdi32 user32)
    if(CSV_CURVE_WIN7)
        if(NOT MSVC OR MSVC_VERSION GREATER_EQUAL 1950)
            message(FATAL_ERROR "CSV_CURVE_WIN7 requires MSVC v143 (14.44) or an earlier Windows 7-capable toolset.")
        endif()
        target_compile_definitions(csv_curve PRIVATE _WIN32_WINNT=0x0601 WINVER=0x0601)
        target_link_options(csv_curve PRIVATE /SUBSYSTEM:WINDOWS,6.01 /OSVERSION:6.1)
    endif()
    if(MSVC AND CMAKE_SIZEOF_VOID_P EQUAL 4)
        target_link_options(csv_curve PRIVATE /LARGEADDRESSAWARE)
    endif()
    if(MINGW)
        target_link_options(csv_curve PRIVATE -municode)
        target_link_options(convert PRIVATE -municode)
    endif()
endif()
